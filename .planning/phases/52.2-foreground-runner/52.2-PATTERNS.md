```rust
pub fn probe(path: &Path) -> std::io::Result<Held> {
    let file = match File::open(path) {
        Ok(file) => file,
        Err(error) if error.kind() == std::io::ErrorKind::NotFound => return Ok(Held::Missing),
        Err(error) => return Err(error),

        Ok(()) => Ok(Held::Abandoned),
        Err(TryLockError::WouldBlock) => Ok(Held::Live),
        Err(TryLockError::Error(error)) => Err(error),
    }
}
```

```rust
pub fn install_signal_pipe() -> io::Result<()> {
    static INSTALLED: OnceLock<Result<(), (io::ErrorKind, String)>> = OnceLock::new();
    INSTALLED
        .get_or_init(|| install_once().map_err(|error| (error.kind(), error.to_string())))
        .clone()
        .map_err(|(kind, message)| io::Error::new(kind, message))
}
fn install_once() -> io::Result<()> {
    let [read, write] = cloexec_nonblocking_pipe()?;
    SIGNAL_WRITE_FD.store(write, Ordering::SeqCst);
    SUPERVISOR_PID.store(unsafe { libc::getpid() }, Ordering::SeqCst);
    SIGNAL_READ_FD.store(read, Ordering::SeqCst);
    for signal in [libc::SIGTERM, libc::SIGINT] { /* zeroed sigaction, SA_RESTART, sigemptyset, sigaction */ }
    Ok(())
}
```

```rust
pub(crate) fn path_digest(path: &Path) -> u64 {
    use std::os::unix::ffi::OsStrExt;
    const FNV_OFFSET_BASIS: u64 = 0xcbf2_9ce4_8422_2325;
    const FNV_PRIME: u64 = 0x0000_0100_0000_01b3;
    let mut hash = FNV_OFFSET_BASIS;
    for byte in path.as_os_str().as_bytes() { hash ^= u64::from(*byte); hash = hash.wrapping_mul(FNV_PRIME); }
    hash
}
```

### Error reporting
**Source:** `crates/devflow-cli/src/commands.rs` `claim_foreground_holder` L4880-4915

`CliError::Message(format!("phase {phase}: <what> ({err}); <verb to run>"))` — name the phase, the cause, and the one

**Source:** `commands.rs` `discard` L1425-1440
**Apply to:** `start`, `resume`, `discard`, `cleanup` (per phase): `claim_foreground_holder` → state read under
`update_state_with_within(…, WRITE_LOCK_REPAIR_WAIT, …)` → one liveness function → D-07a wait → act. `start`/`resume`
release the holder only after `systemd-run` returns 0.

**Apply to:** `.devflow/.runner-NN` and `.devflow/.holder-NN`: never removed, replaced or `unlock()`ed; excluded from
every cleanup/discard deletion list.

**Source:** `git.rs` `hermetic_command_at` L96-100; `monitor.rs` `strip_notify_env` L403
**Apply to:** systemd-run/systemctl invocations, the agent `Command` (`env_remove("NOTIFY_SOCKET")`), and all tests.
